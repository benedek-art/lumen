import Foundation
import LumenCore

// ---------- helpers ----------
let toSRGB = RGBColorSpace.rec2020.matrix(to: .srgb)

func image(_ vals: [RGB]) -> ImageBuffer {
    var img = ImageBuffer(width: vals.count, height: 1)
    for (i, v) in vals.enumerated() { img[i, 0] = v }
    return img
}
func renderVals(_ vals: [RGB], _ r: Recipe, lut: Int = 33) -> [RGB] {
    let plan = RenderPlan(recipe: r, asShotKelvin: 6503.7072, asShotTint: 9.77, lutSize: lut)
    let out = ReferenceRenderer.render(image(vals), plan: plan)
    return (0..<vals.count).map { out[$0, 0] }
}
func exactVals(_ vals: [RGB], _ r: Recipe) -> [RGB] {
    let plan = RenderPlan(recipe: r, asShotKelvin: 6503.7072, asShotTint: 9.77)
    return vals.map { plan.exactColor($0, space: .rec2020) }
}
func srgbEnc(_ x: Double) -> Double { TransferFunction.srgb.encode(Swift.max(0, Swift.min(1, x))) }
func codes(_ c: RGB) -> RGB { let s = toSRGB.apply(c); return RGB(srgbEnc(s.r) * 255, srgbEnc(s.g) * 255, srgbEnc(s.b) * 255) }
func cbrt_(_ x: Double) -> Double { x < 0 ? -pow(-x, 1.0 / 3) : pow(x, 1.0 / 3) }
/// OKLab from linear sRGB.
func oklab(_ c: RGB) -> (L: Double, C: Double, h: Double) {
    let l = 0.4122214708 * c.r + 0.5363325363 * c.g + 0.0514459929 * c.b
    let m = 0.2119034982 * c.r + 0.6806995451 * c.g + 0.1073969566 * c.b
    let s = 0.0883024619 * c.r + 0.2817188376 * c.g + 0.6299787005 * c.b
    let l_ = cbrt_(l), m_ = cbrt_(m), s_ = cbrt_(s)
    let L = 0.2104542553 * l_ + 0.7936177850 * m_ - 0.0040720468 * s_
    let a = 1.9779984951 * l_ - 2.4285922050 * m_ + 0.4505937099 * s_
    let b = 0.0259040371 * l_ + 0.7827717662 * m_ - 0.8086757660 * s_
    var h = atan2(b, a) * 180 / .pi; if h < 0 { h += 360 }
    return (L, hypot(a, b), h)
}
func shipped(_ c: RGB) -> RGB { let s = toSRGB.apply(c); return RGB(Swift.max(0, Swift.min(1, s.r)), Swift.max(0, Swift.min(1, s.g)), Swift.max(0, Swift.min(1, s.b))) }
func hd(_ a: Double, _ b: Double) -> Double { var d = a - b; while d > 180 { d -= 360 }; while d < -180 { d += 360 }; return d }
func ev(_ k: Double) -> Recipe { var r = Recipe(); r.develop.tone.exposure = k; return r }
func f(_ x: Double, _ p: Int = 3) -> String { String(format: "%.\(p)f", x) }

let mode = CommandLine.arguments.dropFirst().first ?? "all"

// ---------- A: S6 is exactly 2^EV ----------
if mode == "all" || mode == "s6" {
    print("== A. S6 matrix gain ==")
    let probes: [Double] = [-1, -0.01, -1e-6, 0, 1e-6, 0.001, 0.18, 1, 5.76, 64, 1000, 65504]
    for wb in [nil, 3200.0, 9000.0] as [Double?] {
        for pl in [0, 3] {
            var base = Recipe(); base.develop.raw.temp = wb; base.develop.raw.tint = wb == nil ? nil : 5
            base.look.printerLights = PrinterLights(master: 0, r: pl, g: 0, b: -pl)
            let p0 = RenderPlan(recipe: base, asShotKelvin: 5000, asShotTint: 0)
            var worst = 0.0
            for k in [-10.0, -5, -3, -1, 1, 2, 3, 5, 10] {
                var r = base; r.develop.tone.exposure = k
                let pk = RenderPlan(recipe: r, asShotKelvin: 5000, asShotTint: 0)
                for i in 0..<3 { for j in 0..<3 {
                    let a = p0.linear.matrix.m[i][j] * pow(2, k), b = pk.linear.matrix.m[i][j]
                    if a != 0 || b != 0 { worst = Swift.max(worst, abs(a - b) / Swift.max(abs(a), 1e-300)) }
                } }
                for v in probes {
                    let c = RGB(v, v * 0.5, -v * 0.3)
                    let o0 = p0.linear.apply(c), ok = pk.linear.apply(c)
                    for (x, y) in [(o0.r, ok.r), (o0.g, ok.g), (o0.b, ok.b)] where x != 0 {
                        worst = Swift.max(worst, abs(y / (x * pow(2, k)) - 1))
                    }
                }
            }
            print("wb=\(wb.map { String($0) } ?? "asShot") printerLights r=\(pl): max rel err of M(EV)=2^EV*M(0) over EV in ±10, probes -1..65504: \(worst)")
        }
    }
    // Float storage (ImageBuffer is Float32): exactness through the buffer, and out-of-range clamp
    let vals: [Double] = [-1, -0.001, 0.001, 0.18, 1, 64, 1000]
    for k in [1.0, -1.0, 10, 11, -11] {
        let r = ev(k)
        let plan = RenderPlan(recipe: r)
        let img = image(vals.map { RGB(gray: $0) })
        let out = img.map { plan.linear.matrix.apply($0) }
        let ratios = (0..<vals.count).map { out[$0, 0].g / vals[$0] }
        print("EV \(k): buffer ratios out/in =", ratios.map { f($0, 6) }.joined(separator: " "), " exposureGain=\(plan.tone.exposureGain)")
    }
}

// ---------- B: pure-gain equivalence end to end: render(R+EVk, x) == render(R, 2^k x) ----------
if mode == "all" || mode == "equiv" {
    print("== B. render(R with EV=k, x) vs render(R, 2^k x) ==")
    var ramp: [RGB] = []
    for i in 0..<400 {
        let e = -10 + 20 * Double(i) / 399
        let v = 0.18 * pow(2, e)
        ramp.append([RGB(v, v, v), RGB(v * 0.3, v * 0.6, v * 1.9), RGB(v * 2.2, v * 1.3, v * 0.8), RGB(v * 1.5, v * 0.2, v * 0.1)][i % 4])
    }
    ramp.append(RGB(-0.02, 0.01, 0.05)); ramp.append(RGB(0, 0, 0)); ramp.append(RGB(100, 90, 80))
    var recipes: [(String, Recipe)] = []
    recipes.append(("default", Recipe()))
    var r = Recipe(); r.develop.tone.contrast = 40; r.develop.tone.highlights = -60; r.develop.tone.shadows = 50; r.develop.tone.whites = 30; r.develop.tone.blacks = -20; recipes.append(("tone stack", r))
    r = Recipe(); r.develop.zones.mid.ev = 0.5; r.develop.zones.bright.ev = -0.7; recipes.append(("zones", r))
    r = Recipe(); r.develop.color.saturation = 40; r.develop.color.vibrance = 30; recipes.append(("sat/vib", r))
    r = Recipe(); r.look.wheels.high = Wheel(hue: 40, sat: 0.4, lum: 0.2); r.look.wheels.shadows = Wheel(hue: 220, sat: 0.3, lum: -0.1); recipes.append(("grade wheels", r))
    r = Recipe(); r.look.render.preset = "Punchy"; recipes.append(("Punchy", r))
    r = Recipe(); r.look.render.preset = "Linear"; recipes.append(("Linear", r))
    r = Recipe(); r.look.vignette = -50; recipes.append(("vignette", r))
    r = Recipe(); r.develop.detail.clarity = 40; r.develop.detail.dehaze = 30; recipes.append(("clarity+dehaze", r))
    r = Recipe(); r.look.filmLab = FilmLab(stock: "lumen/portra400"); recipes.append(("film portra", r))
    r = Recipe(); r.develop.raw.temp = 4200; r.develop.raw.tint = 8; recipes.append(("WB 4200", r))
    for (name, base) in recipes {
        var worst = 0.0
        for k in [-2.0, 1, 3] {
            var rk = base; rk.develop.tone.exposure = k
            let a = renderVals(ramp, rk)
            let b = renderVals(ramp.map { $0 * pow(2, k) }, base)
            for i in 0..<ramp.count {
                let ca = codes(a[i]), cb = codes(b[i])
                worst = Swift.max(worst, abs(ca.r - cb.r), abs(ca.g - cb.g), abs(ca.b - cb.b))
            }
        }
        print("\(name): max |code diff| over EV {-2,1,3}: \(f(worst, 6))")
    }
}

// ---------- C: mid-grey placement ----------

    print("== C. 0.18 grey placement (sRGB code, LUT33 / exact) ==")
    for k in -3...3 {

        print("EV \(k): code(0.18 @EV) = \(f(codes(a).g, 2))  exact \(f(codes(e).g, 2))  code(0.18*2^k @EV0) = \(f(codes(b).g, 2))  display-linear Y=\(f(a.g, 4))")
    }
    // Neutral ramp codes at EV 0 for reference stops
    let stops: [Double] = [-9, -8, -6, -4, -2, -1, 0, 1, 2, 3, 4, 5, 6]

}

// ---------- D: monotonicity / smoothness ----------
if mode == "all" || mode == "mono" {
    print("== D. sweep smoothness ==")
    let scene: [Double] = [0.001, 0.01, 0.05, 0.18, 0.5, 1, 2, 4, 8, 64]
    let neutralRGB = scene.map { RGB(gray: $0) }
    var evs: [Double] = []
    var e = -5.0; while e <= 5.0001 { evs.append((e * 100).rounded() / 100); e += 0.01 }
    for (label, lut) in [("LUT33", 33), ("LUT65", 65), ("exact", 0)] {
        var table: [[Double]] = []
        for k in evs {
            let out = lut == 0 ? exactVals(neutralRGB, ev(k)) : renderVals(neutralRGB, ev(k), lut: lut)
            table.append(out.map { codes($0).g })
        }
        print("-- \(label)")
        for (j, s) in scene.enumerated() {
            var nonMono = 0, maxStep = 0.0, maxStepAt = 0.0, maxCurv = 0.0, curvAt = 0.0
            var deadStart: Double? = nil; var dead: [String] = []
            for i in 1..<evs.count {
                let d = table[i][j] - table[i - 1][j]
                if d < -1e-9 { nonMono += 1 }
                if d > maxStep { maxStep = d; maxStepAt = evs[i] }
                if i >= 2 {
                    let d0 = table[i - 1][j] - table[i - 2][j]
                    let c = abs(d - d0)
                    if c > maxCurv && table[i][j] < 254.9 && table[i-2][j] > 0.1 { maxCurv = c; curvAt = evs[i - 1] }
                }
                let flat = abs(d) < 1e-6 && table[i][j] < 254.999 && table[i][j] > 0.001
                if flat && deadStart == nil { deadStart = evs[i - 1] }
                if !flat, let ds = deadStart { if evs[i - 1] - ds > 0.05 { dead.append("\(ds)..\(evs[i-1])") }; deadStart = nil }
            }
            let clipAt = evs.first { table[evs.firstIndex(of: $0)!][j] >= 254.999 }
            print("scene \(s): nonMono=\(nonMono) maxStep/0.01EV=\(f(maxStep, 3)) @\(maxStepAt)  max|Δ²|=\(f(maxCurv, 4)) @\(curvAt)  firstClip255@EV=\(clipAt.map { String($0) } ?? "-")  dead(within 0<c<255): \(dead.isEmpty ? "none" : dead.joined(separator: ","))")
        }
        // around 0
        let i0 = evs.firstIndex(of: 0)!
        print("grey codes EV -0.03..+0.03:", (i0 - 3...i0 + 3).map { f(table[$0][3], 3) }.joined(separator: " "))
    }
}

// ---------- E: hue / chroma drift vs EV ----------
if mode == "all" || mode == "hue" {
    print("== E. OKLab hue/chroma of shipped sRGB output vs EV (LUT33 pipeline; exact in brackets) ==")
    let patches: [(String, RGB)] = [
        ("sky blue", RGB(0.05, 0.12, 0.35)),
        ("skin", RGB(0.45, 0.30, 0.22)),
        ("dark skin", RGB(0.20, 0.12, 0.08)),
        ("sat red", RGB(0.45, 0.04, 0.03)),
        ("orange", RGB(0.55, 0.22, 0.03)),
        ("sat green", RGB(0.06, 0.30, 0.05)),
        ("foliage", RGB(0.08, 0.14, 0.04)),
        ("deep blue", RGB(0.02, 0.03, 0.30)),
    ]
    let evs: [Double] = [-3, -2, -1, 0, 0.5, 1, 1.5, 2, 2.5, 3, 3.5, 4]
    for (name, c) in patches {
        let sceneLab = oklab(toSRGB.apply(c))
        print("-- \(name) scene rec2020 \(c.r),\(c.g),\(c.b): scene OKLab h=\(f(sceneLab.h, 1)) C=\(f(sceneLab.C, 3))")
        var ref: (Double, Double)? = nil
        var line = ""
        for k in evs {
            let o = renderVals([c], ev(k))[0]
            let x = exactVals([c], ev(k))[0]
            let lab = oklab(shipped(o)), labx = oklab(shipped(x))
            let labu = oklab(toSRGB.apply(x))  // unclamped display-linear
            if k == 0 { ref = (lab.h, lab.C) }
            let cd = codes(o)
            line += "  EV\(k): code(\(Int(cd.r.rounded())),\(Int(cd.g.rounded())),\(Int(cd.b.rounded()))) L=\(f(lab.L, 3)) C=\(f(lab.C, 3)) h=\(f(lab.h, 1)) [exact h=\(f(labx.h, 1)) unclamped h=\(f(labu.h, 1)) C=\(f(labu.C,3))] vsScene dh=\(f(hd(lab.h, sceneLab.h), 1))\n"
        }
        print(line, terminator: "")
        _ = ref
    }
}
